def prepare(provider: str, session_id: str, scope: str = 'latest_prompt_cycle') -> dict:
    row = find_thread(provider, session_id)
    allowed = ('latest_prompt_cycle', 'full_thread_concise', 'full_thread')
    last_n = re.fullmatch(r'last_n_turns:([1-9][0-9]?)', scope)
    if scope not in allowed and not last_n:
        raise ValueError('Choose latest response, full compact, full detailed, or 1–99 recent turns.')
    with _render_lock:
        path = Path(row['session_file'])
        turns = trace.parse_claude_session(path) if row['provider'] == 'claude_code' else trace.parse_codex_session(path)
        if not turns and row['provider'] == 'codex':
            turns = trace.codex_native_selection_turns(path, session_id)
        if not turns:
            raise ValueError('This chat has no readable turns yet.')
        if last_n:
            selected = trace.merge_prompt_cycle(turns[-int(last_n[1]):], threshold_words=0)
        else:
            selected = trace.select_trace_window(turns, turn_arg=None, active=True, allow_partial=True,
                prompt_cycle=scope == 'latest_prompt_cycle', full_thread=scope.startswith('full_thread'),
                threshold_words=trace.SHORT_PROMPT_CHAIN_WORD_THRESHOLD)
        title = row.get('title') or f"{row['provider']} {session_id[:8]}"
        native = trace.render_codex_native_trace(selected, source_window=scope, detail=scope != 'full_thread_concise')
        native_metadata = {}
        if native is not None:
            body, native_metadata = native
            # Reuse the exact snapshots whose items produced the text. Reading
            # the live source again here could bind the text to a later item.
            snapshots = [trace.read_codex_native_snapshot(ref['sha256'], directory=SUPPORT / 'MCP Traces')
                         for ref in native_metadata['native_snapshot_refs']]
            tree = {'schema':'agent_trace_native_tree_v1','session_id':session_id,'snapshots':snapshots,
                    'links':native_metadata['native_links'],
                    'unavailable_children':native_metadata['native_unavailable_children']}
            snapshot_digest = save_native_snapshot(tree, include_children=scope.startswith('full_thread'))
            native_metadata = {**native_metadata, 'snapshot_sha256': snapshot_digest}
        elif scope == 'full_thread_concise':
            body = trace.render_thread_closeout_report(selected, title=title, include_linked_child_compacts=True)
        else:
            body, _ = trace.render_trace_capsule_text(selected, title=title, source_window=scope)
        raw = body.encode()
        if len(raw) > MAX_TRACE_BYTES:
            raise ValueError('This trace is larger than 16 MB. Choose Full thread compact or fewer recent turns.')
        digest = hashlib.sha256(raw).hexdigest()
        folder = SUPPORT / 'MCP Traces'
        folder.mkdir(parents=True, exist_ok=True)
        destination = folder / (digest + '.md')
        if not destination.exists():
            temp = folder / (digest + '.tmp')
            temp.write_bytes(raw)
            temp.replace(destination)
        label = {'full_thread_concise':'full thread compact','full_thread':'full thread detailed','latest_prompt_cycle':'latest response'}.get(scope, scope.replace('last_n_turns:', 'last ') + ' turns')
        resource = {'type':'resource_link', 'uri':f'agent-trace://artifact/{digest}', 'name':f'{title} — {label}',
                    'mimeType':'text/markdown', 'size':len(raw), 'description':f"{row['provider']} · {session_id} · {label}"}
        return {'ok':True, 'provider':row['provider'], 'session_id':session_id, 'scope':scope,
                'title':title, 'bytes':len(raw), 'sha256':digest, 'resource':resource, 'text':body,
                'turn_index':selected.turn_index, **{k:v for k,v in native_metadata.items() if k not in ('raw_sidecar_text','raw_sidecars')}}


    @server.tool(title='Search native trace items', annotations=cached_write)
    async def search_trace_items(provider: str, session_id: str, query: str = '', kind: str = '',
                                 failures_only: bool = False, include_children: bool = False,
                                 snapshot_sha256: str = '', limit: int = 20, offset: int = 0) -> CallToolResult:
        """Search recorded native items, including outputs/diffs. Reuse snapshot_sha256 to page immutable results."""
        data = await asyncio.to_thread(search_items, provider, session_id, query, kind, failures_only,
                                       include_children, snapshot_sha256, limit, offset)
        return CallToolResult(content=[TextContent(type='text', text=json.dumps(data, ensure_ascii=False))], structuredContent=data)


    @server.tool(title='Read exact native trace item', annotations=cached_write)
    async def read_trace_item(provider: str, session_id: str, item_id: str, snapshot_sha256: str = '',
                              offset: int = 0, limit: int = 24000, field: str = '') -> CallToolResult:
        """Read one exact item, or a JSON Pointer field relative to its native item body (for example /aggregated_output). Empty field reads the whole item. Use its own session_id for child items; page with the returned immutable snapshot."""
        data = await asyncio.to_thread(read_item, provider, session_id, item_id, snapshot_sha256, offset, limit, field)
        body = data.pop('payload_json', None)
        if body is None:
            body = json.dumps(data.pop('payload'), ensure_ascii=False)
        return CallToolResult(content=[TextContent(type='text', text=body)], structuredContent=data)


    @server.tool(title='Read native trace changes', annotations=cached_write)
    async def read_trace_changes(provider: str, session_id: str, since_sha256: str,
                                 include_children: bool = False) -> CallToolResult:
        """Compare a prior snapshot with current native IDs; report added, updated and removed items."""
        data = await asyncio.to_thread(read_changes, provider, session_id, since_sha256, include_children)
        return CallToolResult(content=[TextContent(type='text', text=json.dumps(data, ensure_ascii=False))], structuredContent=data)
